cmake_minimum_required(VERSION 3.24)
project(ScientificComputingEngine VERSION 0.1.0 LANGUAGES CXX)
add_library(sce_lu src/lu.cpp src/solve.cpp)
target_include_directories(sce_lu PUBLIC include)
target_compile_features(sce_lu PUBLIC cxx_std_20)
add_library(sce_qr src/qr.cpp)
target_include_directories(sce_qr PUBLIC include)
target_compile_features(sce_qr PUBLIC cxx_std_20)
set_target_properties(sce_qr PROPERTIES CXX_EXTENSIONS OFF)
set_target_properties(sce_lu PROPERTIES CXX_EXTENSIONS OFF)
add_executable(lu_probe tools/lu_probe.cpp)
target_link_libraries(lu_probe PRIVATE sce_lu)
if(MSVC)
  target_compile_options(sce_lu PRIVATE /W4 /fp:strict)
  target_compile_options(sce_qr PRIVATE /W4 /fp:strict)
else()
  target_compile_options(sce_lu PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
  target_compile_options(sce_qr PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
endif()
add_executable(solve_probe tools/solve_probe.cpp)
target_link_libraries(solve_probe PRIVATE sce_lu)
add_executable(record_example tools/record_example.cpp)
target_link_libraries(record_example PRIVATE sce_lu)
include(CTest)
if(BUILD_TESTING)
  find_package(GTest CONFIG QUIET)
  if(NOT GTest_FOUND)
    set(SCE_GTEST_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/googletest-1.17.0" CACHE PATH "Existing GoogleTest source; never downloaded automatically")
    if(NOT EXISTS "${SCE_GTEST_SOURCE}/CMakeLists.txt")
      message(FATAL_ERROR "Provide GTest or set SCE_GTEST_SOURCE; use -DBUILD_TESTING=OFF for library only")
    endif()
    set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
    set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
    add_subdirectory("${SCE_GTEST_SOURCE}" "${CMAKE_BINARY_DIR}/googletest" EXCLUDE_FROM_ALL)
  endif()
  add_executable(lu_tests tests/lu_tests.cpp tests/solve_tests.cpp)
  add_executable(qr_tests tests/qr_tests.cpp)
  target_link_libraries(qr_tests PRIVATE sce_qr GTest::gtest_main)
  target_link_libraries(lu_tests PRIVATE sce_lu GTest::gtest_main)
  include(GoogleTest)
  gtest_discover_tests(lu_tests)
  gtest_discover_tests(qr_tests)
  set(SCE_REFERENCE_PYTHON "" CACHE FILEPATH "Optional Python with existing NumPy")
  if(SCE_REFERENCE_PYTHON)
    add_test(NAME Solve.NumPyReference COMMAND "${SCE_REFERENCE_PYTHON}" -B
      "${CMAKE_CURRENT_SOURCE_DIR}/tests/solve_reference.py" --probe $<TARGET_FILE:solve_probe>)
    add_test(NAME Lu.NumPyReference COMMAND "${SCE_REFERENCE_PYTHON}" -B
      "${CMAKE_CURRENT_SOURCE_DIR}/tests/numpy_reference.py" --probe $<TARGET_FILE:lu_probe>)
  endif()
endif()
